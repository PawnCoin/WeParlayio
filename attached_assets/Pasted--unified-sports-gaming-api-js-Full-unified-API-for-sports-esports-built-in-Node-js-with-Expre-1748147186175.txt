// unified_sports_gaming_api.js
// Full unified API for sports + esports built in Node.js with Express with API key protection, rate limiting, and caching

const express = require('express');
const axios = require('axios');
const rateLimit = require('express-rate-limit');
const NodeCache = require('node-cache');
require('dotenv').config();

const app = express();
const PORT = process.env.PORT || 3000;

// === API Key middleware ===
const validateApiKey = (req, res, next) => {
  const key = req.headers['x-api-key'];
  if (!key || key !== process.env.MY_API_KEY) {
    return res.status(401).json({ error: 'Unauthorized – Invalid API Key' });
  }
  next();
};

// === Rate Limiting ===
const limiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 100,
  message: { error: 'Too many requests, try again later.' }
});

// === Caching ===
const cache = new NodeCache({ stdTTL: 60 }); // cache for 60 seconds

// === Middleware ===
app.use(express.json());
app.use(limiter);
app.use(validateApiKey);

// === Helper with cache ===
const fetchWithCache = async (key, url, config = {}) => {
  if (cache.has(key)) return cache.get(key);
  const { data } = await axios.get(url, config);
  cache.set(key, data);
  return data;
};

// === Sample API keys (replace with your own) ===
const FORTNITE_API_KEY = process.env.FORTNITE_API_KEY;
const TRACKER_API_KEY = process.env.TRACKER_API_KEY;
const RIOT_API_KEY = process.env.RIOT_API_KEY;
const PANDA_API_KEY = process.env.PANDA_API_KEY;

// === Fortnite ===
app.get('/api/esports/fortnite/:username', async (req, res) => {
  try {
    const key = `fortnite-${req.params.username}`;
    const data = await fetchWithCache(key, `https://fortnite-api.com/v2/stats/br/v2?name=${req.params.username}`, {
      headers: { Authorization: FORTNITE_API_KEY }
    });
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: 'Fortnite fetch error' });
  }
});

// === NBA ===
app.get('/api/sports/nba/players', async (req, res) => {
  try {
    const data = await fetchWithCache('nba-players', 'https://www.balldontlie.io/api/v1/players');
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: 'NBA fetch error' });
  }
});

// === College Football ===
app.get('/api/sports/cfb/teams', async (req, res) => {
  try {
    const data = await fetchWithCache('cfb-teams', 'https://api.collegefootballdata.com/teams');
    res.json(data);
  } catch (err) {
    res.status(500).json({ error: 'CFB fetch error' });
  }
});

// === Default Route ===
app.get('/', (req, res) => {
  res.send('🎮 Unified Sports & Gaming API (with Auth & Cache) is running');
});

// === Start Server ===
app.listen(PORT, () => console.log(`✅ API running on port ${PORT}`));
