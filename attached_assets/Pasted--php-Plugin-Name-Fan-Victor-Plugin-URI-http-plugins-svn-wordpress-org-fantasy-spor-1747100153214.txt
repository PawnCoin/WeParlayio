function fanvictorShowTeamsField() {
    $mSports = new Sports();
    $aSports = $mSports->getSports();
    $aSportTeams = array();
    foreach ($aSports as $aSport) {
        if ($aSport['is_team'] != 1 || $aSport['is_active'] != 1 || $aSport['siteID'] != 0) {
            continue;
        }
        $aSport = $aSport['child'];
        foreach ($aSport as $sport) {
            if ($sport['is_team'] != 1 || $sport['is_active'] != 1 || $sport['siteID'] != 0) {
                continue;
            }
            $aSportTeams[$sport["id"]] = $sport["name"];
        }
    }

    if (empty($aSportTeams)) {
        return;
    }
    $fanvictor = new Fanvictor();
    $listSports = $fanvictor->getTeamsBySports(array_keys($aSportTeams));
    return array($listSports, $aSportTeams);
}

function fanvictor_add_registration_fields() {
    require_once(FANVICTOR__PLUGIN_DIR_MODEL . 'admin/sports.php');
    require_once(FANVICTOR__PLUGIN_DIR_MODEL . 'fanvictor.php');
    list($listSports, $aSportTeams) = fanvictorShowTeamsField();
    if (!empty($listSports)) {
        echo '<p><label for="team_info">' . __('Teams', 'fantasy-sports') . '</label>';
        echo '<select  name="team_info" id="team-info" class="input">';
        foreach ($listSports as $sport_id => $sports) {
            echo '<optgroup  label="' . $aSportTeams[$sport_id] . '">';
            foreach ($sports as $item) {
                echo '<option value="' . $sport_id . '_' . $item['teamID'] . '">' . $item['name'] . '</option>';
            }
            echo '</optgroup>';
        }
        echo '</select></p>';
    }
}

function fanvictor_user_register($user_id) {
    if (isset($_POST['team_info']) && !empty($_POST['team_info'])) {
        global $wpdb;
        $tb_user_teams = $wpdb->prefix . 'user_teams';
        $user = $wpdb->get_row("SELECT * FROM $tb_user_teams WHERE user_id=$user_id", ARRAY_A);
        $team_info = $_POST['team_info'];
        $info = explode('_', $team_info);
        $org_id = $info[0];
        $teamID = $info[1];
        if (!empty($user)) { // update
            $id = $user['id'];
            $wpdb->update(
                    $tb_user_teams, array('team_id' => $teamID, 'org_id' => $org_id), array('id' => $id), array('%d', '%d'), array('%d')
            );
        } else { // insert
            $wpdb->insert(
                    $tb_user_teams, array('user_id' => $user_id, 'team_id' => $teamID, 'org_id' => $org_id), array('%d', '%d', '%d')
            );
        }
    }
}

function fv_change_background_by_user() {
    $is_allow = fanvictorCheckAllowChangebackground();
    if (!$is_allow) {
        return false;
    }

    global $wpdb;
    $tbl_user_teams = $wpdb->prefix . 'user_teams';
    $user_id = get_current_user_id();
    $team = $wpdb->get_row("SELECT * FROM $tbl_user_teams WHERE user_id=$user_id", ARRAY_A);
    if (!empty($team)) {
        $f_img_path = FANVICTOR__PLUGIN_DIR . 'assets/teams/' . $team['team_id'];
        $file_info = glob($f_img_path . '.*');
        if (!empty($file_info) && file_exists($file_info[0])) {
            $abs_path_img = $file_info[0];
            $aPath = explode('/', $abs_path_img);
            $file_name = $aPath[count($aPath) - 1];
            $url_background = FANVICTOR__PLUGIN_URL . "assets/teams/$file_name";
            echo "<style>body.fv_background_class{background-image:url('" . $url_background . "') !important;}</style>";
        }
    }
}

function FV_SaveUserProfileField( $user_id ){
    if ( !current_user_can( 'edit_user', $user_id ) ) { 
        return false; 
    }
    global $wpdb;
    $prefix = $wpdb->prefix;
    $table = $prefix . "user_extended";
    $city = $_POST['city'];
	$date_of_birth = date('Y-m-d', strtotime($_POST['date_of_birth']));
    $sql = "UPDATE $table SET city = ' " . $city . " ', date_of_birth = ' " . $date_of_birth . " ' WHERE user_id = $user_id";
    $wpdb->query($sql);
}
    
function FV_UserProfileField( $user ) {
    wp_enqueue_style('ui.css', FANVICTOR__PLUGIN_URL_CSS.'ui/ui.css');
    wp_enqueue_script('ui.js', FANVICTOR__PLUGIN_URL_JS.'ui.js');
    echo '<script>
            jQuery(window).load(function(){
                jQuery( "#date_of_birth" ).datepicker({maxDate: "0", dateFormat: "mm/dd/yy", changeMonth: true});
            })
          </script>';
    
    $fanvictor = new Fanvictor();
    $citys = $fanvictor->getCountryList();
    global $wpdb;
    $prefix = $wpdb->prefix;
    $table = $prefix . "user_extended";
    $user_id = $user->ID;
    $sql = "SELECT * FROM $table WHERE user_id = $user_id ";
    $user_extended = $wpdb->get_row($sql);
    $cur_city = !empty($user_extended->city) ? $user_extended->city : "";
	$date_of_birth = !empty($user_extended->date_of_birth) ? date("m/d/Y", strtotime($user_extended->date_of_birth)) : "";
    echo '<table class="form-table">';
    echo '<tbody>';
    echo '<tr class="user-email-wrap">';
    echo '<th><label for="city">Country</label></th>';
    echo '<td>';
    echo '<select name="city" id="city">';
    foreach($citys as $k => $city){
        $f = ($k == trim($cur_city))?"selected":"";
        echo '<option ' . $f . ' value='. $k .'>' . $city . '</option>';
    }
    echo '</select>';
	echo '</td>';
    echo '</tr>';
	
	echo '<tr class="user-email-wrap">';
    echo '<th><label for="date_of_birth">Date of Birth</label></th>';
    echo '<td>';
    echo '<input type="text" name="date_of_birth" id="date_of_birth" value="'.$date_of_birth.'" />';
	echo '</td>';
    echo '</tr>';
	
    echo '</tbody>';
    echo '</table>';

}
